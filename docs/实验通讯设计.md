# 实验内部通讯设计

## 通讯的函数实现

```javascript
// 实验内容中需要实现的通讯方法示例
window.addEventListener('DOMContentLoaded',()=>{
    // 通知父页面实验已加载
        parent.postMessage({ 
        type:'EXPERIMENT_READY' 
        },'*'
    );

    // 实验步骤完成时发送进度
    functionsendProgress(step){
            parent.postMessage({
            type:'PROGRESS_UPDATE',
            value: step
            },'*');
    }

    // 实验操作记录
    functionrecordOperation(action){
            parent.postMessage({
            type:'OPERATION_RECORD',
            action: action.type,
            timestamp:Date.now(),
            data: action.data
            },'*');
    }

    // 实验完成时调用
    functioncompleteExperiment(score){
            parent.postMessage({
            type:'EXPERIMENT_COMPLETE',
            score: score,
            timestamp:Date.now()
            },'*'
        );
    }
});
```

## 封装为JS库

```javascript
classExperimentCommunicator{
constructor(options ={}){
// 初始化配置参数
this.debug= options.debug||false;
this.onReady= options.onReady||null;
this.onProgress= options.onProgress||null;
this.onComplete= options.onComplete||null;

// 绑定事件监听
window.addEventListener('DOMContentLoaded',()=>this.init());
}

init(){
// 发送实验就绪通知
this.sendMessage('EXPERIMENT_READY');

// 绑定Animate事件监听
if(window.createjs){
this.bindAnimateEvents(); 
}
}

// 通用消息发送方法
sendMessage(type, data ={}){
const message ={ type,...data };
    parent.postMessage(message,'*');
this.debug&&console.log('[Experiment] 发送消息:', message);
}

// 绑定Animate事件监听
bindAnimateEvents(){
// 示例：监听动画完成事件
    createjs.Ticker.addEventListener('tick',(event)=>{
// 这里可以添加动画进度跟踪逻辑
});
}

// 进度更新
updateProgress(step){
this.sendMessage('PROGRESS_UPDATE',{value: step });
}

// 记录操作
recordOperation(actionType, data){
this.sendMessage('OPERATION_RECORD',{
action: actionType,
timestamp:Date.now(),
      data
});
}

// 完成实验
complete(score){
this.sendMessage('EXPERIMENT_COMPLETE',{ score });
}
}

// 暴露全局访问（适配An项目）
window.ExperimentComm=ExperimentCommunicator;
```

使用实例

```javascript
// 初始化通讯器
const comm = new ExperimentComm({
  debug: true,
  onReady: () => {
    // 实验加载完成回调
  },
  onProgress: (step) => {
    // 处理进度更新 
  }
});

// 在动画关键帧调用（示例）
function onStepComplete(step) {
  comm.updateProgress(step);
}

// 在按钮点击事件中
function onUserAction() {
  comm.recordOperation('BUTTON_CLICK', { 
    target: 'startButton' 
  });
}
```
