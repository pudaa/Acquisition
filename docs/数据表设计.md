# 数据表设计

```sql
-- 用户表
CREATE TABLE `users` (
    `id` int NOT NULL AUTO_INCREMENT,
    `username` varchar(50) COLLATE utf8mb4_bin NOT NULL COMMENT '学生学号',
    `password` varchar(255) COLLATE utf8mb4_bin NOT NULL,
    `created_at` timestamp NULL DEFAULT CURRENT_TIMESTAMP,
    `updated_at` timestamp NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    `role` varchar(20) COLLATE utf8mb4_bin NOT NULL DEFAULT 'student',
    `avatar_url` varchar(255) COLLATE utf8mb4_bin DEFAULT '/default_avatar.png' COMMENT '头像存储路径',
    `realname` varchar(255) COLLATE utf8mb4_bin NOT NULL COMMENT '学生姓名',
    PRIMARY KEY (`id`),
    UNIQUE KEY `username` (`username`),
    UNIQUE KEY `email_UNIQUE` (`realname`)
) ENGINE = InnoDB AUTO_INCREMENT = 5 DEFAULT CHARSET = utf8mb4 COLLATE = utf8mb4_bin

-- 实验主表
CREATE TABLE experiments (
    exp_id INT PRIMARY KEY AUTO_INCREMENT,
    title VARCHAR(100) NOT NULL,      -- 实验标题
    subject VARCHAR(50) NOT NULL DEFAULT '物理',     -- 实验所属学科，默认物理
    difficulty ENUM('basic', 'intermediate', 'advanced') NOT NULL, -- 难度等级
    cover_url VARCHAR(200),           -- 封面图片URL，封面路径位置
    duration SMALLINT NOT NULL,      -- 预计完成时间(分钟)←不重要，可以随便填
    steps JSON NOT NULL               -- 实验步骤（目标）配置(JSON格式)
);

-- 用户-实验关联表（跟踪学习进度）
CREATE TABLE user_experiments (
    ue_id INT PRIMARY KEY AUTO_INCREMENT,  -- 主键ID
    user_id INT NOT NULL,            -- 关联的用户ID
    exp_id INT NOT NULL,             -- 关联的实验ID
    progress FLOAT DEFAULT 0,        -- 学习进度(0-100%)
    is_completed BOOLEAN DEFAULT 0,   -- 是否完成(0未完成/1已完成)
    last_studied DATETIME,           -- 最后学习时间
    UNIQUE (user_id, exp_id)         -- 确保每个用户每个实验只记录一次
);

-- 用户操作进度表
CREATE TABLE experiment_attempts (
    attempt_id INT PRIMARY KEY AUTO_INCREMENT, -- 当前学习活动id
    user_id INT NOT NULL, --学生id
    exp_id INT NOT NULL, --实验id
    start_time DATETIME DEFAULT CURRENT_TIMESTAMP, -- 记录时间（这两个时间目前是不准确有用的）
    end_time DATETIME, --结束时间
    progress FLOAT DEFAULT 0, --进度（0——100的数字）
    is_completed BOOLEAN DEFAULT 0, --是否完成
    operations JSON, -- 操作，是一个列表，里面包含了完成了的目标清单
    goals JSON, --包含了各个目标，用true和false来标记是否完成
    FOREIGN KEY (user_id) REFERENCES users(id), 
    FOREIGN KEY (exp_id) REFERENCES experiments(exp_id)
);

-- 讨论话题表
CREATE TABLE discussion_topics (
    id INT PRIMARY KEY AUTO_INCREMENT,
    title VARCHAR(255) NOT NULL,           -- 话题标题
    content TEXT NOT NULL,                 -- 话题内容
    author_id INT NOT NULL,                -- 作者用户ID
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP, -- 创建时间
    updated_at DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP, -- 更新时间
    is_pinned BOOLEAN DEFAULT 0,           -- 是否置顶
    FOREIGN KEY (author_id) REFERENCES users(id)
);

-- 讨论回复表
CREATE TABLE discussion_replies (
    id INT PRIMARY KEY AUTO_INCREMENT,
    topic_id INT NOT NULL,                 -- 所属话题ID
    content TEXT NOT NULL,                 -- 回复内容
    author_id INT NOT NULL,                -- 回复用户ID
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP, -- 回复时间
    FOREIGN KEY (topic_id) REFERENCES discussion_topics(id) ON DELETE CASCADE,
    FOREIGN KEY (author_id) REFERENCES users(id)
);

-- 题库表
CREATE TABLE questions (
    id INT PRIMARY KEY AUTO_INCREMENT,
    exp_id INT NOT NULL,                    -- 关联的实验ID
    title TEXT NOT NULL,                    -- 题目内容
    type ENUM('single', 'multiple') NOT NULL, -- 题目类型：单选/多选
    difficulty ENUM('high', 'medium', 'low') NOT NULL, -- 难度等级
    weight INT NOT NULL,                    -- 权重分值(10/7/3)
    options JSON NOT NULL,                  -- 选项数组
    correct_answers JSON NOT NULL,          -- 正确答案
    knowledge_point VARCHAR(255),           -- 知识点
    FOREIGN KEY (exp_id) REFERENCES experiments(exp_id)
);

-- 学生练习记录表
CREATE TABLE practice_records (
    id INT PRIMARY KEY AUTO_INCREMENT,
    user_id INT NOT NULL,
    exp_id INT NOT NULL,
    question_id INT NOT NULL,
    user_answer JSON NOT NULL,              -- 学生的答案
    is_correct BOOLEAN NOT NULL,            -- 是否正确
    difficulty ENUM('high', 'medium', 'low') NOT NULL,
    score INT NOT NULL,                     -- 该题得分
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(id),
    FOREIGN KEY (exp_id) REFERENCES experiments(exp_id),
    FOREIGN KEY (question_id) REFERENCES questions(id)
);

-- 练习会话表(记录一次完整的10题练习)
CREATE TABLE practice_sessions (
    id INT PRIMARY KEY AUTO_INCREMENT,
    user_id INT NOT NULL,
    exp_id INT NOT NULL,
    total_score INT NOT NULL,               -- 总分
    mastery_score FLOAT NOT NULL,           -- 最终掌握度
    completed_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(id),
    FOREIGN KEY (exp_id) REFERENCES experiments(exp_id)
);
```
